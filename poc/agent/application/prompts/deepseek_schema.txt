    "target_apps": [{"app_id": "稳定小写ID", "app_name": "App名称"}],

  },
  "constraints": ["用户明确约束"],
  "completion_conditions": [{
    "condition_id": "稳定小写ID",
    "description": "最终需要出现的正向结果",
    "evidence_required": ["从动作后新画面或结果中需要看到的事实"]
  }],
  "effect_intents": [{
    "effect_id": "稳定小写ID",
    "kind": "send_message|publish_content|relationship_change|membership_change|data_mutation|authentication|financial_transaction|sensitive_permission_change|irreversible_account_deletion|irreversible_data_deletion",
    "target_entity_roles": ["goal.entities 中的对象角色"],

expected_results。本地运行时会从计划顺序、依赖、当前新截图和一次动作结果派生这些显示字段。
